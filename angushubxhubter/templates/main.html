📜 Roblox Executor Script

คัดลอกสคริปนี้ไปรันใน executor — จะสแกนคลังไอเทมทั้งหมดพร้อมดึงรูปภาพจากเกม แล้วส่งมาแสดงบนเว็บ

-- Roblox Inventory Scanner v3.0
-- รองรับ Blox Fruits (Backpack + Storage + Weapons)
-- ดึง TextureId และจัดหมวดหมู่อัตโนมัติ

local HttpService = game:GetService("HttpService")
local Players = game:GetService("Players")
local MarketplaceService = game:GetService("MarketplaceService")
local LP = Players.LocalPlayer
local SERVER_URL = "http://localhost:5000"

local function extractAssetId(tex)
    if not tex or tex == "" then return nil end
    return tex:match("(%d+)")
end

local function scanItem(item)
    local info = {
        name=item.Name, className=item.ClassName,
        textureId="", assetId=nil, toolTip="", category="Other"
    }
    if item:IsA("Tool") then
        info.textureId = item.TextureId or ""
        info.assetId = extractAssetId(info.textureId)
        info.toolTip = item.ToolTip or ""
    end
    if not info.assetId then
        for _, child in pairs(item:GetDescendants()) do
            if child:IsA("Decal") and child.Texture ~= "" then
                info.assetId = extractAssetId(child.Texture); info.textureId = child.Texture; break
            elseif child:IsA("ImageLabel") and child.Image ~= "" then
                info.assetId = extractAssetId(child.Image); info.textureId = child.Image; break
            end
        end
    end
    return info
end

local function categorize(item)
    local n = item.name:lower()
    local t = (item.toolTip or ""):lower()
    local fk = {"fruit","rubber","flame","ice","light","dark","magma","quake","buddha","phoenix","dragon","dough","venom","control","spirit","leopard","kitsune","mammoth","t-rex","rumble","blizzard","shadow","portal","sound","pain","gas"}
    local sk = {"sword","blade","katana","cutlass","saber","trident","pole","bisento","yoru","shisui","wando","tushita","yama","scythe"}
    local gk = {"gun","pistol","rifle","musket","cannon","flintlock","kabucha","guitar","bow"}
    local fstyle = {"combat","fighting style","superhuman","death step","sharkman","electric claw","godhuman","sanguine"}
    local acc = {"cape","coat","hat","mask","bandana","accessory"}
    for _, k in ipairs(fk) do if n:find(k) or t:find(k) then return "Fruit" end end
    for _, k in ipairs(sk) do if n:find(k) or t:find(k) then return "Sword" end end
    for _, k in ipairs(gk) do if n:find(k) or t:find(k) then return "Gun" end end
    for _, k in ipairs(fstyle) do if n:find(k) or t:find(k) then return "Fighting Style" end end
    for _, k in ipairs(acc) do if n:find(k) or t:find(k) then return "Accessory" end end
    if item.className == "Tool" then return "Item" end
    return "Other"
end

local inv = {}
for _,item in pairs(LP.Backpack:GetChildren()) do
    if item:IsA("Tool") then
        local i = scanItem(item); i.category = categorize(i); i.equipped = false; i.source = "Backpack"
        table.insert(inv, i)
    end
end
if LP.Character then
    for _,item in pairs(LP.Character:GetChildren()) do
        if item:IsA("Tool") then
            local i = scanItem(item); i.category = categorize(i); i.equipped = true; i.source = "Equipped"
            table.insert(inv, i)
        end
    end
end

-- Blox Fruits Fruit/Weapon Storage
pcall(function()
    local comm = game:GetService("ReplicatedStorage"):FindFirstChild("Remotes") and game:GetService("ReplicatedStorage").Remotes:FindFirstChild("CommF_")
    if comm then
        local f = comm:InvokeServer("getInventoryFruits")
        if type(f) == "table" then
            for _, item in pairs(f) do
                local fn = type(item) == "table" and (item.Name or item.name) or tostring(item)
                table.insert(inv, {name=fn, className="StoredFruit", category="Fruit", equipped=false, source="Fruit Storage"})
            end
        end
    end
end)

local gName = "Unknown"
pcall(function() gName = MarketplaceService:GetProductInfo(game.PlaceId).Name end)

local payload = {
    user_id = tostring(LP.UserId), username = LP.Name, display_name = LP.DisplayName,
    place_id = tostring(game.PlaceId), game_name = gName,
    inventory = inv, game_items = {}, scan_time = os.date("%Y-%m-%d %H:%M:%S")
}

local ok, res = pcall(function()
    return request({
        Url = SERVER_URL .. "/api/inventory", Method = "POST",
        Headers = {["Content-Type"] = "application/json"},
        Body = HttpService:JSONEncode(payload)
    })
end)

if ok then
    print("✅ ส่งข้อมูลแล้ว! พบ " .. #inv .. " ชิ้น")
else
    warn("❌ Error: " .. tostring(res))
end
