📜 Roblox Executor Script

คัดลอกสคริปนี้ไปรันใน Roblox executor เพื่อส่งข้อมูลผู้เล่นมาที่เว็บ

-- Roblox Player Tracker Script
-- รันสคริปนี้ใน executor (Synapse X, Fluxus, etc.)

local HttpService = game:GetService("HttpService")
local Players = game:GetService("Players")
local LocalPlayer = Players.LocalPlayer

-- ⚠️ เปลี่ยน URL นี้เป็น IP ของคุณ (หรือใช้ localhost ถ้ารันบนเครื่องเดียวกัน)
local SERVER_URL = "http://localhost:5000/api/track"

local data = {
    user_id = tostring(LocalPlayer.UserId),
    username = LocalPlayer.Name,
    display_name = LocalPlayer.DisplayName
}

local jsonData = HttpService:JSONEncode(data)

local success, response = pcall(function()
    return request({
        Url = SERVER_URL,
        Method = "POST",
        Headers = {
            ["Content-Type"] = "application/json"
        },
        Body = jsonData
    })
end)

if success then
    print("✅ ส่งข้อมูลสำเร็จ!")
    print("📊 Response: " .. tostring(response.Body))
else
    print("❌ ส่งข้อมูลไม่สำเร็จ: " .. tostring(response))
end
